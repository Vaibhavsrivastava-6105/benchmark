"use client";

import { 
  Play, 
  ChevronRight, 
  ChevronLeft, 
  Server,

const API_BASE = "http://localhost:8000";

interface Provider {
  id: number;
  name: string;
  type: string;
  base_url: string;
  enabled: boolean;
  status?: string;
  models?: string[];
}

interface PromptSuite {
  id: number;
  name: string;
  description: string;
}

      try {
        // Fetch Providers
        const provRes = await fetch(`${API_BASE}/api/providers`);
        const provs = await provRes.json();
        
        // Load Provider statuses in background
        const activeProvs = provs.filter((p: Provider) => p.enabled);
        setProviders(activeProvs);
        
        // Fetch Prompt Suites
        const suiteRes = await fetch(`${API_BASE}/api/prompts`);

        console.error("Failed to load wizard sources:", err);

        setLoadingProviders(false);

      const res = await fetch(`${API_BASE}/api/providers/${id}/health`, { method: "POST" });

        if (p.id === id) {
          return { ...p, status: data.status, models: data.models };

        return p;

          return { ...p, status: "OFFLINE" };

        return p;

        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)

        const runData = await res.json();
        router.push(`/benchmarks/${runData.id}`);

        const err = await res.json();

            }`}>

                {s.num}

        {/* STEP 1: MODEL SELECTION */}

          <div className="space-y-6">
            <h3 className="text-lg font-bold text-zinc-200 flex items-center gap-2">
              <Settings className="h-5 w-5 text-cyan-500" />
              Configure Target Model

            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              <div className="space-y-1.5">
                <label className="text-xs uppercase font-bold tracking-wider text-zinc-500">Benchmark Run Name</label>

                  type="text"
