"use client";

import React, { useState, useEffect } from "react";
import { Cpu, Plus, HelpCircle, HardDrive } from "lucide-react";

const API_BASE = "http://localhost:8000";

interface Model {
  id: number;
  name: string;
  revision?: string;
  quantization?: string;
  size_bytes?: number;
  context_length?: number;
  parameters?: string;
  architecture?: string;
}

export default function ModelsPage() {
  const [models, setModels] = useState<Model[]>([]);
  const [loading, setLoading] = useState(true);

  // Form
  const [showForm, setShowForm] = useState(false);
  const [name, setName] = useState("");
  const [revision, setRevision] = useState("latest");
  const [quantization, setQuantization] = useState("FP16");
  const [sizeBytes, setSizeBytes] = useState<number>(0);
  const [contextLength, setContextLength] = useState<number>(8192);
  const [parameters, setParameters] = useState("8B");
  const [architecture, setArchitecture] = useState("Llama");

  const fetchModels = async () => {
    try {
      const res = await fetch(`${API_BASE}/api/models`);
      const data = await res.json();
      setModels(data);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchModels();
  }, []);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!name) return;

    const payload = {
      name,
      revision,
      quantization,
      size_bytes: sizeBytes || null,
      context_length: contextLength || null,
      parameters,
      architecture
    };

    try {
      const res = await fetch(`${API_BASE}/api/models`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      });
      if (res.ok) {
        setShowForm(false);
        setName("");
        fetchModels();
      } else {
        alert("Failed to register model.");
      }
    } catch (err) {
      console.error(err);
    }
  };

  return (
    <div className="p-8 space-y-8 flex-1">
      {/* Header */}
      <div className="flex justify-between items-center">
        <div>
          <h1 className="text-3xl font-bold tracking-tight">Models Library</h1>
          <p className="text-zinc-400 text-sm mt-1">Manage and register target LLM models for execution comparison.</p>
        </div>
        <button
          onClick={() => setShowForm(!showForm)}
          className="flex items-center gap-2 px-4 py-1.5 rounded-lg text-sm font-semibold bg-cyan-500 text-zinc-950 hover:bg-cyan-400 transition-colors"
        >
          <Plus className="h-4 w-4 text-zinc-950" />
          Register Model
        </button>
      </div>

      {/* Form */}
      {showForm && (
        <form onSubmit={handleSubmit} className="bg-[#0c0c0e] border border-zinc-800 rounded-xl p-6 max-w-xl space-y-4">
          <h3 className="font-bold text-zinc-200">Register Target Model Parameters</h3>
          
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div className="space-y-1.5 md:col-span-2">
              <label className="text-xs uppercase font-bold tracking-wider text-zinc-500">Model Name / Hugging Face ID</label>
              <input
                type="text"
                placeholder="e.g. meta-llama/Meta-Llama-3-8B-Instruct"
                value={name}
                onChange={(e) => setName(e.target.value)}
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg p-2 text-sm font-mono focus:outline-none focus:border-cyan-500"
              />
            </div>
            
            <div className="space-y-1.5">
              <label className="text-xs uppercase font-bold tracking-wider text-zinc-500">Architecture</label>
              <input
                type="text"
                placeholder="e.g. Llama3"
                value={architecture}
                onChange={(e) => setArchitecture(e.target.value)}
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg p-2 text-sm focus:outline-none focus:border-cyan-500"
              />
            </div>

            <div className="space-y-1.5">
              <label className="text-xs uppercase font-bold tracking-wider text-zinc-500">Parameters Size</label>
              <input
                type="text"
                placeholder="e.g. 8B"
                value={parameters}
                onChange={(e) => setParameters(e.target.value)}
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg p-2 text-sm focus:outline-none focus:border-cyan-500"
              />
            </div>

            <div className="space-y-1.5">
              <label className="text-xs uppercase font-bold tracking-wider text-zinc-500">Quantization</label>
              <input
                type="text"
                placeholder="e.g. FP16, Q4_K_M"
                value={quantization}
                onChange={(e) => setQuantization(e.target.value)}
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg p-2 text-sm focus:outline-none focus:border-cyan-500"
              />
            </div>

            <div className="space-y-1.5">
              <label className="text-xs uppercase font-bold tracking-wider text-zinc-500">Context Window</label>
              <input
                type="number"
                value={contextLength}
                onChange={(e) => setContextLength(parseInt(e.target.value))}
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg p-2 text-sm focus:outline-none focus:border-cyan-500"
              />
            </div>
          </div>

          <div className="flex justify-end gap-3 pt-4">
            <button
              type="button"
              onClick={() => setShowForm(false)}
              className="px-4 py-2 border border-zinc-800 text-zinc-400 hover:text-white rounded-lg text-sm transition-colors"
            >
              Cancel
            </button>
            <button
              type="submit"
              className="px-5 py-2 rounded-lg text-sm font-bold bg-cyan-500 text-zinc-950 hover:bg-cyan-400 transition-colors"
            >
              Register
            </button>
          </div>
        </form>
      )}

      {/* Models list */}
      <div className="bg-[#0c0c0e] border border-zinc-800 rounded-xl p-6">
        {loading ? (
          <div className="text-zinc-500 font-mono text-xs">Querying database models...</div>
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm border-collapse">
              <thead>
                <tr className="text-zinc-400 border-b border-zinc-800 font-semibold text-xs uppercase tracking-wider">
                  <th className="py-3 px-4">Model ID</th>
                  <th className="py-3 px-4">Model Name</th>
                  <th className="py-3 px-4">Architecture</th>
                  <th className="py-3 px-4">Params</th>
                  <th className="py-3 px-4">Quantization</th>
                  <th className="py-3 px-4">Context Length</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-zinc-850 font-mono text-xs text-zinc-300">
                {models.map((m) => (
                  <tr key={m.id} className="hover:bg-zinc-900/40 transition-colors">
                    <td className="py-3.5 px-4 text-zinc-500">#{m.id}</td>
                    <td className="py-3.5 px-4 font-sans font-semibold text-zinc-200">{m.name}</td>
                    <td className="py-3.5 px-4 text-zinc-400">{m.architecture || "Unknown"}</td>
                    <td className="py-3.5 px-4 text-cyan-400 font-semibold">{m.parameters || "N/A"}</td>
                    <td className="py-3.5 px-4 text-zinc-400">
                      <span className="px-2 py-0.5 rounded bg-zinc-900 border border-zinc-800 uppercase text-[10px]">
                        {m.quantization || "FP16"}
                      </span>
                    </td>
                    <td className="py-3.5 px-4 text-zinc-400">{m.context_length ? `${m.context_length.toLocaleString()} tokens` : "N/A"}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
    </div>
  );
}
